```python
def predict_rapidminer_tree(p: dict) -> dict:
    """
    Eksekusi langsung pohon keputusan RapidMiner untuk deteksi Alzheimer.
    
    Parameter:
        p (dict): Fitur pasien, contoh:
            {
                "ADL": 4.5,
                "CholesterolLDL": 120.0,
                "MMSE": 18.0,
                "FunctionalAssessment": 6.2,
                "MemoryComplaints": 1,
                "BehavioralProblems": 1,
                "CholesterolHDL": 55.0,
                "Ethnicity": 0,
                "SystolicBP": 135.0,
                "CholesterolTriglycerides": 180.0,
                "PhysicalActivity": 2.5,
                "AlcoholConsumption": 8.0,
                "CholesterolTotal": 210.0,
                "Age": 72,
                "BMI": 26.5
            }
            
    Return:
        dict: diagnosis (0/1), label, confidence (probabilitas), leaf_counts, rule_path
    """
    adl = float(p.get("ADL", 0.0))

    # Cabang Utama 1: ADL <= 0.012
    if adl <= 0.012:
        return {
            "diagnosis": 1,
            "label": "Alzheimer",
            "confidence": 3 / 3,
            "leaf_counts": {"0": 0, "1": 3},
            "rule_path": "ADL <= 0.012"
        }

    # Cabang Utama 2: ADL > 0.012
    ldl = float(p.get("CholesterolLDL", 0.0))
    if ldl > 199.872:
        return {
            "diagnosis": 1,
            "label": "Alzheimer",
            "confidence": 2 / 2,
            "leaf_counts": {"0": 0, "1": 2},
            "rule_path": "ADL > 0.012 -> CholesterolLDL > 199.872"
        }

    # CholesterolLDL <= 199.872
    mmse = float(p.get("MMSE", 0.0))
    if mmse > 24.010:
        fa = float(p.get("FunctionalAssessment", 0.0))
        if fa > 9.894:
            return {
                "diagnosis": 1,
                "label": "Alzheimer",
                "confidence": 2 / 3,
                "leaf_counts": {"0": 1, "1": 2},
                "rule_path": "MMSE > 24.010 -> FunctionalAssessment > 9.894"
            }
        else:
            return {
                "diagnosis": 0,
                "label": "Non-Alzheimer",
                "confidence": 394 / 410,
                "leaf_counts": {"0": 394, "1": 16},
                "rule_path": "MMSE > 24.010 -> FunctionalAssessment <= 9.894"
            }

    # MMSE <= 24.010
    fa = float(p.get("FunctionalAssessment", 0.0))
    if fa > 5.001:
        mc = int(p.get("MemoryComplaints", 0))
        if mc == 0:
            bp = int(p.get("BehavioralProblems", 0))
            if bp == 0:
                return {
                    "diagnosis": 0,
                    "label": "Non-Alzheimer",
                    "confidence": 585 / 605,
                    "leaf_counts": {"0": 585, "1": 20},
                    "rule_path": "MMSE <= 24.010 -> FA > 5.001 -> MemoryComplaints=0 -> BehavioralProblems=0"
                }
            else:  # BehavioralProblems == 1
                if adl > 5.038:
                    hdl = float(p.get("CholesterolHDL", 0.0))
                    if hdl > 93.227:
                        eth = float(p.get("Ethnicity", 0.0))
                        if eth > 1.500:
                            return {
                                "diagnosis": 0,
                                "label": "Non-Alzheimer",
                                "confidence": 2 / 2,
                                "leaf_counts": {"0": 2, "1": 0},
                                "rule_path": "ADL > 5.038 -> HDL > 93.227 -> Ethnicity > 1.500"
                            }
                        else:
                            return {
                                "diagnosis": 1,
                                "label": "Alzheimer",
                                "confidence": 4 / 5,
                                "leaf_counts": {"0": 1, "1": 4},
                                "rule_path": "ADL > 5.038 -> HDL > 93.227 -> Ethnicity <= 1.500"
                            }
                    else:
                        return {
                            "diagnosis": 0,
                            "label": "Non-Alzheimer",
                            "confidence": 54 / 55,
                            "leaf_counts": {"0": 54, "1": 1},
                            "rule_path": "ADL > 5.038 -> HDL <= 93.227"
                        }
                else:  # ADL <= 5.038
                    sbp = float(p.get("SystolicBP", 0.0))
                    if sbp > 152.500:
                        trig = float(p.get("CholesterolTriglycerides", 0.0))
                        if trig > 273.179:
                            return {
                                "diagnosis": 0,
                                "label": "Non-Alzheimer",
                                "confidence": 3 / 3,
                                "leaf_counts": {"0": 3, "1": 0},
                                "rule_path": "ADL <= 5.038 -> SystolicBP > 152.500 -> Triglycerides > 273.179"
                            }
                        else:
                            return {
                                "diagnosis": 1,
                                "label": "Alzheimer",
                                "confidence": 7 / 8,
                                "leaf_counts": {"0": 1, "1": 7},
                                "rule_path": "ADL <= 5.038 -> SystolicBP > 152.500 -> Triglycerides <= 273.179"
                            }
                    else:
                        return {
                            "diagnosis": 1,
                            "label": "Alzheimer",
                            "confidence": 34 / 35,
                            "leaf_counts": {"0": 1, "1": 34},
                            "rule_path": "ADL <= 5.038 -> SystolicBP <= 152.500"
                        }
        else:  # MemoryComplaints == 1
            if adl > 4.989:
                bp = int(p.get("BehavioralProblems", 0))
                if bp == 0:
                    return {
                        "diagnosis": 0,
                        "label": "Non-Alzheimer",
                        "confidence": 68 / 72,
                        "leaf_counts": {"0": 68, "1": 4},
                        "rule_path": "MemoryComplaints=1 -> ADL > 4.989 -> BehavioralProblems=0"
                    }
                else:
                    return {
                        "diagnosis": 1,
                        "label": "Alzheimer",
                        "confidence": 17 / 17,
                        "leaf_counts": {"0": 0, "1": 17},
                        "rule_path": "MemoryComplaints=1 -> ADL > 4.989 -> BehavioralProblems=1"
                    }
            else:  # ADL <= 4.989
                pa = float(p.get("PhysicalActivity", 0.0))
                if pa > 3.171:
                    return {
                        "diagnosis": 1,
                        "label": "Alzheimer",
                        "confidence": 59 / 59,
                        "leaf_counts": {"0": 0, "1": 59},
                        "rule_path": "MemoryComplaints=1 -> ADL <= 4.989 -> PhysicalActivity > 3.171"
                    }
                else:
                    alc = float(p.get("AlcoholConsumption", 0.0))
                    if alc > 15.338:
                        return {
                            "diagnosis": 0,
                            "label": "Non-Alzheimer",
                            "confidence": 4 / 5,
                            "leaf_counts": {"0": 4, "1": 1},
                            "rule_path": "PhysicalActivity <= 3.171 -> AlcoholConsumption > 15.338"
                        }
                    else:
                        return {
                            "diagnosis": 1,
                            "label": "Alzheimer",
                            "confidence": 18 / 20,
                            "leaf_counts": {"0": 2, "1": 18},
                            "rule_path": "PhysicalActivity <= 3.171 -> AlcoholConsumption <= 15.338"
                        }
    else:  # FunctionalAssessment <= 5.001
        if adl > 5.040:
            mc = int(p.get("MemoryComplaints", 0))
            if mc == 0:
                bp = int(p.get("BehavioralProblems", 0))
                if bp == 0:
                    return {
                        "diagnosis": 0,
                        "label": "Non-Alzheimer",
                        "confidence": 249 / 257,
                        "leaf_counts": {"0": 249, "1": 8},
                        "rule_path": "FA <= 5.001 -> ADL > 5.040 -> MemoryComplaints=0 -> BehavioralProblems=0"
                    }
                else:
                    chol = float(p.get("CholesterolTotal", 0.0))
                    if chol > 164.786:
                        return {
                            "diagnosis": 1,
                            "label": "Alzheimer",
                            "confidence": 47 / 48,
                            "leaf_counts": {"0": 1, "1": 47},
                            "rule_path": "CholesterolTotal > 164.786"
                        }
                    else:
                        age = float(p.get("Age", 0))
                        if age > 66:
                            return {
                                "diagnosis": 1,
                                "label": "Alzheimer",
                                "confidence": 5 / 5,
                                "leaf_counts": {"0": 0, "1": 5},
                                "rule_path": "Age > 66"
                            }
                        else:
                            return {
                                "diagnosis": 0,
                                "label": "Non-Alzheimer",
                                "confidence": 2 / 2,
                                "leaf_counts": {"0": 2, "1": 0},
                                "rule_path": "Age <= 66"
                            }
            else:  # MemoryComplaints == 1
                return {
                    "diagnosis": 1,
                    "label": "Alzheimer",
                    "confidence": 80 / 82,
                    "leaf_counts": {"0": 2, "1": 80},
                    "rule_path": "FA <= 5.001 -> ADL > 5.040 -> MemoryComplaints=1"
                }
        else:  # ADL <= 5.040
            bmi = float(p.get("BMI", 0.0))
            if bmi > 39.925:
                return {
                    "diagnosis": 0,
                    "label": "Non-Alzheimer",
                    "confidence": 1 / 2,
                    "leaf_counts": {"0": 1, "1": 1},
                    "rule_path": "FA <= 5.001 -> ADL <= 5.040 -> BMI > 39.925"
                }
            else:
                return {
                    "diagnosis": 1,

                    "confidence": 431 / 449,
                    "leaf_counts": {"0": 18, "1": 431},
                    "rule_path": "FA <= 5.001 -> ADL <= 5.040 -> BMI <= 39.925"
```

Skrip ini melatih model XGBoost Classifier dengan fitur yang sama dan
menghitung akurasi, sensitivitas, spesifisitas, ROC-AUC, serta feature importance.

```python
import os
import json
import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.model_selection import train_test_split
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    roc_auc_score,
    confusion_matrix,
    classification_report
)

def train_xgboost_pipeline(
    dataset_csv: str = "datasets/alzheimers_disease_data.csv",
    save_model_path: str = "public/checkpoints/xgboost_alzheimers.json"
):
    print("=" * 65)
    print("PROSES PELATIHAN MODEL XGBOOST ALZHEIMER")
    print("=" * 65)

    if not os.path.exists(dataset_csv):
        raise FileNotFoundError(f"File dataset tidak ditemukan: {dataset_csv}")

    # 1. Baca data
    df = pd.read_csv(dataset_csv)
    print(f"Total baris data: {len(df):,} baris")

    # 2. Hapus identifier yang tidak relevan untuk diagnosis
    drop_cols = ["PatientID", "DoctorInCharge"]
    df_clean = df.drop(columns=[col for col in drop_cols if col in df.columns])

    # 3. Pisahkan Fitur (X) dan Target (y)
    target_col = "Diagnosis"
    X = df_clean.drop(columns=[target_col])
    y = df_clean[target_col]

    # 4. Tangani Class Imbalance
    neg_count = int((y == 0).sum())
    pos_count = int((y == 1).sum())
    scale_pos_weight = neg_count / pos_count
    print(f"Distribusi Kelas: 0 = {neg_count:,} pasien | 1 = {pos_count:,} pasien")
    print(f"Scale Pos Weight: {scale_pos_weight:.2f}")

    # 5. Bagi Data Train & Test (80% Train, 20% Test)
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.20, random_state=42, stratify=y
    )

    # 6. Inisialisasi XGBoost Classifier dengan Regularisasi Optimal
    model = xgb.XGBClassifier(
        n_estimators=120,               # Jumlah decision trees dalam ensemble
        learning_rate=0.08,             # Learning rate (shrinkage)
        max_depth=4,                    # Kedalaman maksimal tiap pohon
        min_child_weight=2,             # Mencegah overfit pada sampel kecil
        gamma=0.1,                      # Minimum loss reduction untuk split
        subsample=0.85,                 # Fraksi data untuk tiap pohon
        colsample_bytree=0.85,          # Fraksi fitur per pohon
        scale_pos_weight=scale_pos_weight,
        eval_metric="logloss",
        use_label_encoder=False,
        random_state=42
    )

    # 7. Pelatihan
    model.fit(
        X_train, y_train,
        eval_set=[(X_train, y_train), (X_test, y_test)],
        verbose=False
    )

    # 8. Prediksi & Evaluasi
    y_pred = model.predict(X_test)
    y_prob = model.predict_proba(X_test)[:, 1]

    acc = accuracy_score(y_test, y_pred)
    prec = precision_score(y_test, y_pred)
    rec = recall_score(y_test, y_pred)
    f1 = f1_score(y_test, y_pred)
    auc = roc_auc_score(y_test, y_prob)
    cm = confusion_matrix(y_test, y_pred)

    print("\n" + "=" * 65)
    print("HASIL METRIK EVALUASI:")
    print("=" * 65)
    print(f"Akurasi           : {acc * 100:.2f}%")
    print(f"Precision         : {prec * 100:.2f}%")
    print(f"Recall/Sensitivity: {rec * 100:.2f}%")
    print(f"F1-Score          : {f1 * 100:.2f}%")
    print(f"ROC-AUC Score     : {auc * 100:.2f}%")
    print(f"Confusion Matrix  :\n{cm}")

    print("\n" + classification_report(y_test, y_pred, target_names=["Non-Alzheimer (0)", "Alzheimer (1)"]))

    # 9. Top 10 Feature Importance (Bobot Fitur)
    importances = pd.Series(model.feature_importances_, index=X.columns).sort_values(ascending=False)
    print("=" * 65)
    print("TOP 10 BIOMARKER TERPENTING (FEATURE IMPORTANCE):")
    print("=" * 65)
    for rank, (feat, score) in enumerate(importances.head(10).items(), 1):
        print(f"{rank:2d}. {feat:<25} : {score * 100:.2f}%")

    # 10. Simpan Model ke Format JSON Checkpoint
    os.makedirs(os.path.dirname(save_model_path), exist_ok=True)
    model.save_model(save_model_path)
    print(f"\nModel tersimpan di: {save_model_path}")

    return model, importances
```

```python
def print_xgboost_tree_dump(model, tree_index: int = 0):
    """
    Mengekstrak pohon ke-i dari ensemble XGBoost dalam bentuk teks hierarkis.
    """
    booster = model.get_booster()
    dump_list = booster.get_dump(with_stats=True)
    
    if tree_index >= len(dump_list):
        print(f"Pohon indeks {tree_index} tidak ditemukan. Total pohon: {len(dump_list)}")
        return
        
    print(f"\n=== STRUKTUR POHON XGBOOST (Tree #{tree_index}) ===")
    print(dump_list[tree_index])
```
